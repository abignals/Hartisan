-- =====================================================================
-- Plateforme artisans — schéma initial
-- PostgreSQL 15+
-- =====================================================================

CREATE EXTENSION IF NOT EXISTS postgis;      -- recherche géographique
CREATE EXTENSION IF NOT EXISTS btree_gist;   -- requis pour l'anti-double-booking
CREATE EXTENSION IF NOT EXISTS pgcrypto;     -- gen_random_uuid()

-- =====================================================================
-- 1. UTILISATEURS & AUTH
-- =====================================================================

CREATE TYPE user_role AS ENUM ('client', 'artisan', 'admin');

CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    role            user_role   NOT NULL,
    email           CITEXT      NOT NULL UNIQUE,   -- CITEXT = insensible à la casse
    password_hash   TEXT        NOT NULL,          -- argon2id ou bcrypt
    first_name      TEXT        NOT NULL,
    last_name       TEXT        NOT NULL,
    phone           TEXT,       NOT NULL,
    email_verified_at TIMESTAMPTZ,
    -- suspension immédiate (litige, fraude) sans supprimer le compte
    suspended_at    TIMESTAMPTZ,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Sessions opaques (pas de JWT) : révocation = DELETE, effet immédiat
CREATE TABLE sessions (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- on stocke un HASH du token, jamais le token en clair :
    -- une fuite de la base ne permet pas d'usurper une session
    token_hash    BYTEA       NOT NULL UNIQUE,
    user_id       UUID        NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    user_agent    TEXT,
    ip_address    INET,
    expires_at    TIMESTAMPTZ NOT NULL,
    revoked_at    TIMESTAMPTZ,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_sessions_user      ON sessions(user_id);
CREATE INDEX idx_sessions_expires   ON sessions(expires_at) WHERE revoked_at IS NULL;

-- =====================================================================
-- 2. MÉTIERS & PROFILS ARTISANS
-- =====================================================================

-- Table de référence : permet d'ouvrir de nouveaux métiers sans migration
CREATE TABLE trades (
    id          SMALLSERIAL PRIMARY KEY,
    slug        TEXT NOT NULL UNIQUE,    -- 'plombier', 'electricien'
    label       TEXT NOT NULL,           -- 'Plombier'
    is_active   BOOLEAN NOT NULL DEFAULT true
);

CREATE TYPE artisan_status AS ENUM ('pending', 'approved', 'rejected', 'suspended');

CREATE TABLE artisan_profiles (
    id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id             UUID NOT NULL UNIQUE REFERENCES users(id) ON DELETE CASCADE,

    company_name        TEXT NOT NULL,
    siret               TEXT UNIQUE,
    description         TEXT,

    -- Adresse de base + point géographique (PostGIS)
    address_line        TEXT,
    postal_code         TEXT NOT NULL,
    city                TEXT NOT NULL,
    location            GEOGRAPHY(Point, 4326) NOT NULL,
    -- rayon d'intervention en mètres
    service_radius_m    INTEGER NOT NULL DEFAULT 20000
                        CHECK (service_radius_m BETWEEN 1000 AND 100000),

    -- durée par défaut d'un RDV, sert à générer les créneaux
    default_slot_minutes SMALLINT NOT NULL DEFAULT 60
                        CHECK (default_slot_minutes BETWEEN 15 AND 480),

    -- Validation manuelle par l'admin (assurance décennale, identité)
    status              artisan_status NOT NULL DEFAULT 'pending',
    insurance_doc_url   TEXT,
    insurance_expires_at DATE,
    reviewed_by         UUID REFERENCES users(id),
    reviewed_at         TIMESTAMPTZ,
    rejection_reason    TEXT,

    -- Dénormalisé pour le tri des résultats de recherche (recalculé sur nouvel avis)
    rating_avg          NUMERIC(2,1),
    rating_count        INTEGER NOT NULL DEFAULT 0,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Index géographique : indispensable pour "artisans à moins de X km"
CREATE INDEX idx_artisan_location ON artisan_profiles USING GIST (location);
CREATE INDEX idx_artisan_status   ON artisan_profiles(status) WHERE status = 'approved';

-- Un artisan peut exercer plusieurs métiers
CREATE TABLE artisan_trades (
    artisan_id  UUID     NOT NULL REFERENCES artisan_profiles(id) ON DELETE CASCADE,
    trade_id    SMALLINT NOT NULL REFERENCES trades(id),
    PRIMARY KEY (artisan_id, trade_id)
);

CREATE INDEX idx_artisan_trades_trade ON artisan_trades(trade_id);

-- Photos de réalisations
CREATE TABLE artisan_photos (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    artisan_id  UUID NOT NULL REFERENCES artisan_profiles(id) ON DELETE CASCADE,
    url         TEXT NOT NULL,
    position    SMALLINT NOT NULL DEFAULT 0,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- =====================================================================
-- 3. DISPONIBILITÉS
-- Modèle : règles récurrentes + exceptions (congés, fermetures ponctuelles).
-- On ne stocke PAS des créneaux pré-générés : on les calcule à la volée
-- à partir des règles, moins les exceptions, moins les bookings existants.
-- =====================================================================

CREATE TABLE availability_rules (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    artisan_id   UUID NOT NULL REFERENCES artisan_profiles(id) ON DELETE CASCADE,
    weekday      SMALLINT NOT NULL CHECK (weekday BETWEEN 0 AND 6), -- 0 = dimanche
    start_time   TIME NOT NULL,
    end_time     TIME NOT NULL,
    -- une règle peut être limitée dans le temps (ex. horaires d'été)
    valid_from   DATE,
    valid_until  DATE,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
    CHECK (end_time > start_time)
);

CREATE INDEX idx_availability_artisan ON availability_rules(artisan_id, weekday);

-- Congés / indisponibilités ponctuelles (écrasent les règles récurrentes)
CREATE TABLE availability_exceptions (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    artisan_id  UUID NOT NULL REFERENCES artisan_profiles(id) ON DELETE CASCADE,
    period      TSTZRANGE NOT NULL,
    reason      TEXT,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_exceptions_artisan ON availability_exceptions USING GIST (artisan_id, period);

-- =====================================================================
-- 4. RÉSERVATIONS  ← le module critique
-- =====================================================================

CREATE TYPE booking_status AS ENUM (
    'pending',    -- demandé par le client, en attente de l'artisan
    'confirmed',  -- accepté par l'artisan
    'rejected',   -- refusé par l'artisan
    'cancelled',  -- annulé (par l'un ou l'autre)
    'completed',  -- intervention réalisée → débloque l'avis
    'no_show'     -- absence constatée
);

CREATE TABLE bookings (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    client_id       UUID NOT NULL REFERENCES users(id),
    artisan_id      UUID NOT NULL REFERENCES artisan_profiles(id),
    trade_id        SMALLINT NOT NULL REFERENCES trades(id),

    -- Le créneau comme intervalle : permet la contrainte d'exclusion ci-dessous
    slot            TSTZRANGE NOT NULL,

    status          booking_status NOT NULL DEFAULT 'pending',

    -- Adresse de l'intervention (pas forcément celle du compte client)
    address_line    TEXT NOT NULL,
    postal_code     TEXT NOT NULL,
    city            TEXT NOT NULL,
    location        GEOGRAPHY(Point, 4326),

    description     TEXT,                 -- le besoin décrit par le client
    -- prix annoncé au moment de la prise de RDV : trace écrite anti-litige
    quoted_price_cents INTEGER,

    cancelled_by    UUID REFERENCES users(id),
    cancelled_at    TIMESTAMPTZ,
    cancellation_reason TEXT,

    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now(),

    CHECK (NOT isempty(slot))
);

-- ⚠️ LA contrainte la plus importante du schéma :
-- impossible d'avoir deux RDV actifs qui se chevauchent pour le même artisan.
-- Garanti par la base, même en cas de bug applicatif ou de race condition
-- (deux clients qui cliquent sur le même créneau à la même milliseconde).
ALTER TABLE bookings ADD CONSTRAINT no_overlapping_bookings
    EXCLUDE USING GIST (
        artisan_id WITH =,
        slot       WITH &&
    ) WHERE (status IN ('pending', 'confirmed'));

CREATE INDEX idx_bookings_artisan_slot ON bookings USING GIST (artisan_id, slot);
CREATE INDEX idx_bookings_client       ON bookings(client_id, created_at DESC);
CREATE INDEX idx_bookings_status       ON bookings(status);

-- =====================================================================
-- 5. AVIS
-- =====================================================================

CREATE TABLE reviews (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- 1 avis max par RDV, et uniquement sur un RDV réellement effectué
    booking_id   UUID NOT NULL UNIQUE REFERENCES bookings(id) ON DELETE CASCADE,
    artisan_id   UUID NOT NULL REFERENCES artisan_profiles(id) ON DELETE CASCADE,
    client_id    UUID NOT NULL REFERENCES users(id),
    rating       SMALLINT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment      TEXT,
    -- modération
    is_published BOOLEAN NOT NULL DEFAULT true,
    hidden_reason TEXT,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_reviews_artisan ON reviews(artisan_id) WHERE is_published;

-- =====================================================================
-- 6. LITIGES
-- =====================================================================

CREATE TYPE dispute_category AS ENUM ('no_show', 'late', 'price_disagreement', 'quality', 'other');
CREATE TYPE dispute_status   AS ENUM ('open', 'in_review', 'resolved', 'escalated');

CREATE TABLE disputes (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    booking_id    UUID NOT NULL REFERENCES bookings(id),
    reported_by   UUID NOT NULL REFERENCES users(id),
    category      dispute_category NOT NULL,
    description   TEXT NOT NULL,
    status        dispute_status NOT NULL DEFAULT 'open',
    resolution    TEXT,
    handled_by    UUID REFERENCES users(id),
    resolved_at   TIMESTAMPTZ,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_disputes_status ON disputes(status) WHERE status IN ('open', 'in_review');

-- =====================================================================
-- 7. ABONNEMENTS (Stripe Billing)
-- =====================================================================

CREATE TYPE subscription_status AS ENUM (
    'pilot_free',   -- gratuit phase pilote
    'trialing',
    'active',
    'past_due',
    'canceled'
);

CREATE TABLE subscriptions (
    id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    artisan_id              UUID NOT NULL UNIQUE REFERENCES artisan_profiles(id) ON DELETE CASCADE,
    status                  subscription_status NOT NULL DEFAULT 'pilot_free',

    stripe_customer_id      TEXT UNIQUE,
    stripe_subscription_id  TEXT UNIQUE,
    stripe_price_id         TEXT,

    -- tarif "fondateur" verrouillé : à honorer même après hausse des prix
    is_founder_rate         BOOLEAN NOT NULL DEFAULT false,
    monthly_price_cents     INTEGER,

    current_period_end      TIMESTAMPTZ,
    canceled_at             TIMESTAMPTZ,
    created_at              TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at              TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- =====================================================================
-- 8. NOTIFICATIONS (file d'envoi email/SMS)
-- =====================================================================

CREATE TABLE notifications (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id      UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    channel      TEXT NOT NULL CHECK (channel IN ('email', 'sms')),
    template     TEXT NOT NULL,           -- 'booking_requested', 'booking_confirmed'...
    payload      JSONB NOT NULL DEFAULT '{}',
    sent_at      TIMESTAMPTZ,
    failed_at    TIMESTAMPTZ,
    error        TEXT,
    created_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_notifications_pending ON notifications(created_at)
    WHERE sent_at IS NULL AND failed_at IS NULL;
